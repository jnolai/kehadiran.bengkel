Simpan Terus ke Google Sheet (Cara Paling Pantas):

  1. Buka Google Sheet baru di akaun anda & beri tajuk Kehadiran Bengkel Perolehan 2027.
  2. Klik menu Extensions > Apps Script.
  3. Padam kod sedia ada, salin kod di bawah dan tampal ke dalam Apps Script.
  4. Klik Deploy > New deployment > Pilih 'Web app'.
  5. Setkan Who has access kepada: Anyone (Penting!).
  6. Salin URL Web App dan tampal dalam kotak di bawah:
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Nama Pegawai", "Jawatan/Gred", "Jabatan", "No. Tel (WhatsApp)", "eMail", "Waktu Daftar"]);
    }
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.namaPegawai || "",
      data.jawatanGred || "",
      data.jabatan || "",
      data.noTel || "",
      data.email || "",
      data.time || ""
    ]);
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var results = [];
  for (var i = 1; i < rows.length; i++) {
    results.push({
      namaPegawai: rows[i][1],
      jawatanGred: rows[i][2],
      jabatan: rows[i][3],
      noTel: rows[i][4],
      email: rows[i][5],
      time: rows[i][6] || rows[i][0]
    });
  }
  return ContentService.createTextOutput(JSON.stringify(results)).setMimeType(ContentService.MimeType.JSON);
}
